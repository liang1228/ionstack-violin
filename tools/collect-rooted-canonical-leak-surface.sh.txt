#!/system/bin/sh
# Rooted same-build evidence collector. Run locally on the rooted violin device.
set -u

if [ "$(id -u)" != "0" ] && command -v su >/dev/null 2>&1; then
  echo "[ionstack] requesting root..."
  exec su -c "sh '$0'"
fi

SCRIPT_DIR=$(CDPATH= cd "$(dirname "$0")" && pwd)
STAMP=$(date +%Y%m%d-%H%M%S 2>/dev/null || echo unknown)
OUT="$SCRIPT_DIR/ionstack-canonical-surface-$STAMP"
echo "[ionstack] uid=$(id -u) output=$OUT"
if ! mkdir -p "$OUT"; then
  echo "[ionstack] ERROR: cannot create output directory"
  exit 1
fi

capture() {
  name=$1
  shift
  echo "[ionstack] collecting $name"
  ("$@") >"$OUT/$name" 2>&1 || true
}

capture meta.txt sh -c 'id; getenforce 2>&1; uname -a; getprop ro.build.fingerprint; cat /proc/sys/kernel/random/boot_id'
capture kallsyms.txt cat /proc/kallsyms
capture iomem.txt cat /proc/iomem
capture modules.txt cat /proc/modules
capture vmallocinfo.txt cat /proc/vmallocinfo
capture slabinfo.txt cat /proc/slabinfo
capture mounts.txt cat /proc/mounts
capture config.gz cat /proc/config.gz
capture dev-list.txt ls -la /dev
capture misc-list.txt sh -c 'find /sys/class/misc -maxdepth 2 -type l -o -type f 2>/dev/null | sort'
capture module-sections.txt sh -c 'find /sys/module -path "*/sections/*" -type f -exec sh -c "printf \"%s=\" \"\$1\"; cat \"\$1\"" _ {} \; 2>/dev/null'
capture gpu-tree.txt sh -c 'find /sys/devices/platform -iname "*gpu*" -o -iname "*mali*" 2>/dev/null | sort'
capture gpu-sysfs.txt sh -c 'find /sys/devices/platform -path "*gpu*" -o -path "*mali*" 2>/dev/null | while read p; do [ -f "$p" ] && { printf "===%s===\n" "$p"; cat "$p"; }; done'
capture tracing-kprobes.txt cat /sys/kernel/tracing/kprobe_events
capture tracing-events.txt sh -c 'find /sys/kernel/tracing/events -maxdepth 3 -name format -type f -exec sh -c "echo ===\$1; cat \$1" _ {} \; 2>/dev/null'
capture kcore-stat.txt sh -c 'ls -l /proc/kcore /proc/last_kmsg 2>&1; [ -r /proc/kcore ] && head -c 4096 /proc/kcore | od -An -tx1 || true'

(cd "$OUT" && sha256sum * > SHA256SUMS.txt 2>/dev/null) || true
echo "[ionstack] packing archive..."
tar -czf "$OUT.tar.gz" -C "$SCRIPT_DIR" "$(basename "$OUT")" 2>/dev/null || true
echo "[ionstack] saved: $OUT"
echo "[ionstack] archive: $OUT.tar.gz"
